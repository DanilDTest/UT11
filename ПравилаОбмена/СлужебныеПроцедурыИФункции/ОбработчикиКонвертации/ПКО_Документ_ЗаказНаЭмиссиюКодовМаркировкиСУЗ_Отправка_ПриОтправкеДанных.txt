Процедура ПКО_Документ_ЗаказНаЭмиссиюКодовМаркировкиСУЗ_Отправка_ПриОтправкеДанных(ДанныеИБ, ДанныеXDTO, КомпонентыОбмена, СтекВыгрузки)
	ТипОснования = ТипЗнч(ДанныеИБ.ДокументОснование); 
	Если ТипОснования = Тип("ДокументСсылка.ЗаказПоставщику") Тогда
		ИмяПКО = "Документ_ЗаказПоставщику_Отправка";
	КонецЕсли;
	
	Если НЕ ПустаяСтрока(ИмяПКО) Тогда 
		
		ДанныеXDTO.Вставить("ДокументОснование", Новый Структура("Значение, ИмяПКО", ДанныеИБ.ДокументОснование, ИмяПКО));  
	
	КонецЕсли;
	
	
	СтруктураДополнительныхРеквизитов = Новый Структура;
	СтруктураДополнительныхРеквизитов.Вставить("ИдентификаторПроизводственногоЗаказа", ДанныеИБ.ИдентификаторПроизводственногоЗаказа);
	СтруктураДополнительныхРеквизитов.Вставить("ИдентификаторПроизводственнойЛинии", ДанныеИБ.ИдентификаторПроизводственнойЛинии);
	СтруктураДополнительныхРеквизитов.Вставить("ПроизводственныйОбъектИдентификатор", ДанныеИБ.ПроизводственныйОбъектИдентификатор);
	СтруктураДополнительныхРеквизитов.Вставить("ПроизводственныйОбъектАдрес", ДанныеИБ.ПроизводственныйОбъектАдрес);
	СтруктураДополнительныхРеквизитов.Вставить("ПроизводственныйОбъектАдресСтрокой", ДанныеИБ.ПроизводственныйОбъектАдресСтрокой);
	СтруктураДополнительныхРеквизитов.Вставить("ОжидаемаяДатаНачалаПроизводства", ДанныеИБ.ОжидаемаяДатаНачалаПроизводства);
	СтруктураДополнительныхРеквизитов.Вставить("ИдентификаторЗаявки", ДанныеИБ.ИдентификаторЗаявки);
	СтруктураДополнительныхРеквизитов.Вставить("УдалитьНомерДоговораСОператором", ДанныеИБ.УдалитьНомерДоговораСОператором);
	СтруктураДополнительныхРеквизитов.Вставить("УдалитьДатаДоговораСОператором", ДанныеИБ.УдалитьДатаДоговораСОператором);
	
	ЗаполнитьЗначенияСвойств(СтруктураДополнительныхРеквизитов, ДанныеИБ);             
	ВыгрузитьДополнительныеРеквизитыСвойствоСтрокой(СтруктураДополнительныхРеквизитов, ДанныеXDTO);
КонецПроцедуры
