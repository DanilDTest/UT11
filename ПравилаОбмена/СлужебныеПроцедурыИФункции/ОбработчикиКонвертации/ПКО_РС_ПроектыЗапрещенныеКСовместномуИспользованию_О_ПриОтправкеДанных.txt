Процедура ПКО_РС_ПроектыЗапрещенныеКСовместномуИспользованию_О_ПриОтправкеДанных(ДанныеИБ, ДанныеXDTO, КомпонентыОбмена, СтекВыгрузки)
	ДанныеXDTO.Вставить("Проект", ДанныеИБ.Отбор.Проект.Значение);    
	
	ТипДокумента = ДанныеИБ.Отбор.ТипДокумента.Значение; 
	ЕстьОтборПоТипу = ЗначениеЗаполнено(ТипДокумента); 
	Если ЕстьОтборПоТипу Тогда  
		 
		ПолноеИмяТипа = ОбщегоНазначения.ЗначениеРеквизитаОбъекта(ТипДокумента, "ПолноеИмя");
		ДанныеXDTO.Вставить("ТипДокумента", ПолноеИмяТипа);     
		
		СоответствиеТипов = Новый Соответствие;
	    СоответствиеТипов.Вставить(ТипДокумента, ПолноеИмяТипа);
	КонецЕсли;
	
	
	ТаблицаЗаписи = ДанныеИБ.Выгрузить(); 
	КолонкаТипСтрока = ТаблицаЗаписи.Колонки.Добавить("ТипДокументаСтрока");
	
	Если ТаблицаЗаписи.Количество() > 0 
		И (НЕ ТаблицаЗаписи.Количество() = 1  и ЕстьОтборПоТипу) Тогда //Если тип 1 то его полное имя уже получили для выгрузки отбора
		
		МассивТипов = ТаблицаЗаписи.ВыгрузитьКолонку("ТипДокумента");
		СоответствиеТипов = ОбщегоНазначения.ЗначениеРеквизитаОбъектов(МассивТипов, "ПолноеИмя"); 	
	КонецЕсли;
	
	Для каждого Запись из ТаблицаЗаписи Цикл 
		Запись.ТипДокументаСтрока = СоответствиеТипов[Запись.ТипДокумента];
	КонецЦикла;       
	
	ТаблицаЗаписи.Колонки.Удалить("ТипДокумента");
	КолонкаТипСтрока.Имя = "ТипДокумента";
	
	ДанныеXDTO.Вставить("Записи", ТаблицаЗаписи);
КонецПроцедуры
