Процедура ПКО_РегистрСведенийЗапись_RM_ИзображенияНоменклатуры_О_ПриОтправкеДанных(ДанныеИБ, ДанныеXDTO, КомпонентыОбмена, СтекВыгрузки)
	ДанныеXDTO.Вставить("Номенклатура", ДанныеИБ.Отбор.Номенклатура.Значение);
	
	Если ДанныеИБ.Отбор.GUIDНоменклатураПрисоединенныеФайлы.Использование Тогда
		УИДИзображения = ДанныеИБ.Отбор.GUIDНоменклатураПрисоединенныеФайлы.Значение;
		ДанныеXDTO.Вставить("GUIDНоменклатураПрисоединенныеФайлы", Строка(УИДИзображения));	
	КонецЕсли;	 
	
	ТаблицаЗаписи = ДанныеИБ.Выгрузить(); 
	ТаблицаЗаписи.Колонки.Добавить("GUIDНоменклатураПрисоединенныеФайлыСтрока");
	Для каждого Стр из ТаблицаЗаписи Цикл
		Стр.GUIDНоменклатураПрисоединенныеФайлыСтрока = Строка(Стр.GUIDНоменклатураПрисоединенныеФайлы);
	КонецЦикла;
	
	ТаблицаЗаписи.Колонки.Удалить("GUIDНоменклатураПрисоединенныеФайлы"); 
	ТаблицаЗаписи.Колонки.GUIDНоменклатураПрисоединенныеФайлыСтрока.Имя = "GUIDНоменклатураПрисоединенныеФайлы";
	
	ДанныеXDTO.Вставить("Записи", ТаблицаЗаписи);
КонецПроцедуры
